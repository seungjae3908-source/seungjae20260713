def replace_once(text: str, old: str, new: str, label: str) -> str:
    if old not in text:
        raise RuntimeError(f"교체 위치를 찾지 못했습니다: {label}")
    return text.replace(old, new, 1)

def regex_once(text: str, pattern: str, replacement: str, label: str) -> str:
    updated, count = re.subn(pattern, replacement, text, count=1, flags=re.S)
    if count != 1:
        raise RuntimeError(f"교체 위치를 찾지 못했습니다: {label}")
    return updated

function coinDirectoryAiGroup(row: AnyObj): CoinDirectoryAiGroup {

function analyzeCoinDetail(candles: AnyObj[], futures: boolean) {
  const closes = candles
    .map((row) => finite(row.close))
    .filter((value): value is number => value != null);
  const volumes = candles
    .map((row) => finite(row.volume))
    .filter((value): value is number => value != null);

  const current = closes.at(-1) ?? null;
  const previous = closes.at(-2) ?? current;
  const sma5 = closes.length
    ? closes.slice(-5).reduce((sum, value) => sum + value, 0) /
      Math.min(5, closes.length)
    : null;
  const sma20 = closes.length
    ? closes.slice(-20).reduce((sum, value) => sum + value, 0) /
      Math.min(20, closes.length)
    : null;

  let gains = 0;
  let losses = 0;
  const rsiRows = closes.slice(-15);

  for (let index = 1; index < rsiRows.length; index += 1) {
    const change = rsiRows[index] - rsiRows[index - 1];
    if (change >= 0) gains += change;
    else losses += Math.abs(change);
  }

  const rsi =
    rsiRows.length < 2
      ? null
      : losses === 0
        ? 100
        : 100 - 100 / (1 + gains / losses);

  const latestVolume = volumes.at(-1) ?? 0;
  const previousVolumes = volumes.slice(-21, -1);
  const averageVolume = previousVolumes.length
    ? previousVolumes.reduce((sum, value) => sum + value, 0) /
      previousVolumes.length
    : 0;
  const volumeRatio =
    averageVolume > 0 ? latestVolume / averageVolume : null;

  let positiveScore = 40;
  let negativeScore = 40;
  const reasons: string[] = [];
  const risks: string[] = [];

  if (current != null && sma5 != null && current > sma5) {
    positiveScore += 15;
    reasons.push('현재가가 단기 이동평균선 위에 있습니다.');
  } else {
    negativeScore += 12;
    risks.push('현재가가 단기 이동평균선 아래에 있습니다.');
  }

  if (sma5 != null && sma20 != null && sma5 > sma20) {
    positiveScore += 17;
    reasons.push('단기 이동평균선이 중기 이동평균선 위에 있어 상승 추세가 우세합니다.');
  } else {
    negativeScore += 14;
    risks.push('단기 이동평균선이 중기 이동평균선 아래에 있어 하락 추세를 확인해야 합니다.');
  }

  if (rsi != null) {
    if (rsi >= 50 && rsi <= 70) {
      positiveScore += 12;
      reasons.push(`RSI ${rsi.toFixed(1)}로 상승 힘이 유지되는 구간입니다.`);
    } else if (rsi >= 75) {
      negativeScore += 13;
      risks.push(`RSI ${rsi.toFixed(1)}로 단기 과열 가능성이 있습니다.`);
    } else if (rsi <= 35) {
      positiveScore += 7;
      reasons.push(`RSI ${rsi.toFixed(1)}로 과매도 반등 가능성을 확인합니다.`);
    }
  }

  if (volumeRatio != null && volumeRatio >= 1.5) {
    if (current != null && previous != null && current >= previous) {
      positiveScore += 13;
      reasons.push(`거래량이 최근 평균의 ${volumeRatio.toFixed(2)}배로 증가했습니다.`);
    } else {
      negativeScore += 13;
      risks.push(`하락 중 거래량이 최근 평균의 ${volumeRatio.toFixed(2)}배로 증가했습니다.`);
    }
  }

  positiveScore = Math.max(0, Math.min(100, Math.round(positiveScore)));
  negativeScore = Math.max(0, Math.min(100, Math.round(negativeScore)));

  const direction =
    positiveScore - negativeScore >= 12
      ? futures
        ? '롱'
        : '매수'
      : negativeScore - positiveScore >= 12
        ? futures
          ? '숏'
          : '매도'
        : '관망';

  const entry = current;
  const stop =
    current == null
      ? null
      : Math.min(sma20 ?? current * 0.97, current * 0.97);
  const target =
    current == null
      ? null
      : current + Math.max(current - (stop ?? current * 0.97), current * 0.015) * 1.8;

  return {
    current,
    sma5,
    sma20,
    rsi,
    volumeRatio,
    positiveScore,
    negativeScore,
    direction,
    reasons,
    risks,
    entry,
    stop,
    target,
  };
}
